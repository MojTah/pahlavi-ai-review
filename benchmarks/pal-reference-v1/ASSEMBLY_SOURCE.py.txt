def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def dump(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")

def lines(path, values):
    path.write_text("".join(json.dumps(x, ensure_ascii=False) + "\n" for x in values), encoding="utf-8")

def norm(text):
    text = unicodedata.normalize("NFC", text).casefold().replace("ي", "ی").replace("ك", "ک")
    return " ".join("".join(c for c in text if c.isalnum() or c.isspace()).split())

CITATION = re.compile(r"\((?P<author>[^()\n,،]+)[,،]\s*(?P<year>(?:[12][0-9]{3}|[۱۲][۰-۹]{3}|n\.d\.))(?P<locator>[^()]*)\)\s*[.،,؛;]*\s*$")
def split_credit(text):
    match = CITATION.search(text)
    if not match:
        raise ValueError("Missing final source citation")
    return text[:match.start()].rstrip(), match.group().strip()

def main():
    if OUT.exists():
        raise SystemExit("Refusing to overwrite existing benchmark")
    data = {s: [json.loads(l) for l in (DATASET / f"{s}-records.jsonl").read_text("utf-8").splitlines()] for s in ("train", "dev", "test")}
    records = {r["id"]: r for r in data["test"]}
    archive = {r["id"]: r for r in map(json.loads, (ARCHIVE / "exports/text-units.jsonl").read_text("utf-8").splitlines())}
    assert sha(ARCHIVE / "exports/text-units.jsonl") == "c42e0a5a0d4d108218a0481ca073c07c6bb00cdcba0bffb462b67bb6d7a60789"
    seen, exposure_files = set(), []
    for path in (TRAINER / "runs").rglob("*"):
        if path.is_file() and path.suffix in (".json", ".jsonl") and (path.name.startswith("predictions") or path.name in ("inputs.json", "responses.json")):
            seen.update(re.findall(r"parsig:\d{9}", path.read_text("utf-8")))
            exposure_files.append({"path": str(path.relative_to(TRAINER)), "sha256": sha(path)})
    selected, items, evidence, collisions = [], [], [], []
    all_prior = [(s, r, norm(r["transcription"])) for s in ("train", "dev") for r in data[s]]
    for work, choices in CHOICES.items():
        for suffixes, checks in choices:
            ids = [f"parsig:{work}000{n:03d}" for n in suffixes]
            assert not set(ids) & seen, ("Previously recorded model exposure", ids)
            segments = []
            for rid in ids:
                r, original = records[rid], archive[rid]
                assert r["work_id"] not in {b["work_id"] for s in ("train", "dev") for b in data[s]}
                raw_texts = {"pal": "\n".join(original["transcription"]), "fa": "\n".join(original["farsi_translation"]), "en": "\n".join(x["text"] for x in original["additional_layers"] if x.get("language") == "en" and x.get("role") == "translation")}
                texts = {"pal": r["transcription"], **{k: r["translations"][k]["text"] for k in ("en", "fa")}}
                credits = {"pal": r["edition"], **{k: r["translations"][k]["credit"] for k in ("en", "fa")}}
                for lang in texts:
                    assert split_credit(raw_texts[lang]) == (texts[lang], credits[lang]), (rid, lang, "source altered")
                response = ARCHIVE / original["source"]["file"]
                assert sha(response) == original["source"]["sha256"]
                ns = norm(texts["pal"])
                for split, other, no in all_prior:
                    if min(len(ns), len(no)) / max(len(ns), len(no)) >= .82:
                        matcher = difflib.SequenceMatcher(None, ns, no, autojunk=False)
                        if matcher.quick_ratio() >= .85 and matcher.ratio() >= .85:
                            collisions.append([rid, split, other["id"], "source", matcher.ratio()])
                    for lang in ("en", "fa"):
                        if lang in other["translations"] and norm(texts[lang]) == norm(other["translations"][lang]["text"]):
                            collisions.append([rid, split, other["id"], lang, 1.0])
                segments.append({"record_id": rid, "texts": texts, "credits": credits, "source": original["source"], "edition_locator": r["locator"]})
                evidence.append(original)
            selected.extend(ids)
            items.append({"id": f"PALREF1-{len(items)+1:03d}", "work_id": "parsig:" + work, "work_title": archive[ids[0]]["book_title"], "record_ids": ids, "texts": {lang: "\n".join(s["texts"][lang] for s in segments) for lang in ("pal", "en", "fa")}, "segments": segments, "meaning_checks": checks, "review_status": "published_references_screened_for_consistency", "expert_adjudicated": False})
    assert len(items) == 40 and len(selected) == len(set(selected)) == 42
    assert not collisions, collisions
    OUT.mkdir(parents=True)
    lines(OUT / "references.jsonl", items)
    lines(OUT / "source-evidence.jsonl", evidence)
    cases = []
    for item in items:
        for source, target in (("pal", "en"), ("pal", "fa"), ("en", "pal"), ("fa", "pal")):
            cases.append({"id": item["id"] + ":" + source + ">" + target, "passage_id": item["id"], "work_id": item["work_id"], "source_language": source, "target_language": target, "source_text": item["texts"][source], "track": "forward_translation" if source == "pal" else "attested_reconstruction"})
    lines(OUT / "inputs.jsonl", cases)
